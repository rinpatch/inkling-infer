```sh
M=$(ls ~/.cache/huggingface/hub/models--ggml-org--Inkling-2-Mini-GGUF/snapshots/*/Inkling-2-Mini-IQ4_NL.gguf)

tools/llama-dump/llama-dump "$M" fixtures/prompts/swa.txt   fixtures/out/swa   32 '^(l_out-(0|4|5|6|11|17|23|30|47)|result_norm|result_output)$' 4096 0 10
tools/llama-dump/llama-dump "$M" fixtures/prompts/long.txt  fixtures/out/long  32 '^(result_norm)$' 512 0 10
```
